4.1 5 行最小流程

#include "Edge10/Edge10_all.h"

Edge10Init(NULL);                       // RC 模式；EP 传 cfg 路径
Edge10Executor* exec = NULL;
Edge10ExecutorCreate(&exec, 0);         // 绑定 device 0
Edge10ExecutorLoadModel(exec, "model.ty");
/* ... 创建输入 Tensor、填数据、Edge10ExecutorInfer(...) ... */
Edge10ExecutorDestroy(exec);
Edge10Finalize();

4.2 完整示例：跑一张随机输入的分类模型

下面这段代码与 samples/src/flow/sample_infer.c 等价，resnet50 输出 1000 类，打印 Top-5：

#include "Edge10/Edge10_all.h"
#include <stdio.h>
#include <string.h>
#include <unistd.h>

static void fill_random(uint8_t* p, size_t n) {
    for (size_t i = 0; i < n; ++i) p[i] = (uint8_t)(rand() & 0xFF);
}

int main(int argc, char** argv) {
    if (argc < 2) { printf("usage: %s <model.ty>\n", argv[0]); return 1; }

    Edge10Error ret = Edge10Init(NULL);          /* RC；EP 传 cfg 路径 */
    if (ret != EDGE10_SUCCESS) return 1;

    Edge10Executor* exec = NULL;
    Edge10ExecutorCreate(&exec, 0);
    Edge10ExecutorLoadModel(exec, argv[1]);

    /* 输入：Host 侧 (1,3,224,224) uint8；device_id = -1 表示 Host */
    const int64_t in_dims[4] = {1, 3, 224, 224};
    Edge10Tensor* in = NULL;
    Edge10TensorCreate(&in, -1, in_dims, 4, EDGE10_DT_UINT8);

    Edge10Buffer* buf = NULL; void* virt = NULL; size_t sz = 0;
    Edge10TensorGetBuffer(in, &buf);
    Edge10GetVirtAddr(buf, &virt);
    Edge10GetBufferSize(buf, &sz);
    fill_random((uint8_t*)virt, sz);

    /* 输出：Host 侧 (1,1000) float */
    const int64_t out_dims[2] = {1, 1000};
    Edge10Tensor* out = NULL;
    Edge10TensorCreate(&out, -1, out_dims, 2, EDGE10_DT_FLOAT);

    /* 推理 */
    ret = Edge10ExecutorInfer(exec, in, out);
    if (ret == EDGE10_SUCCESS) {
        Edge10TensorGetBuffer(out, &buf);
        Edge10GetVirtAddr(buf, &virt);
        const float* logits = (const float*)virt;
        for (int i = 0; i < 5; ++i) printf("class %d : %+.6f\n", i, logits[i]);
    }

    Edge10TensorDestroy(in);
    Edge10TensorDestroy(out);
    Edge10ExecutorDestroy(exec);

    /* EP 模式：让 EP 侧 server 完成资源回收，避免 dclFinalize 时 RPC 竞态 */
    usleep(200 * 1000);
    Edge10Finalize();
    return 0;
}

4.3 运行

SDK 自带预编译好的二进制，直接跑：

source /opt/edge10-sdk/env.sh

# RC（RC NPU）
edge10 --rc flow_infer_rc -m models/onnx_resnet50.ty

# EP（PCIe 远程）
edge10 --ep flow_infer_ep -m models/onnx_resnet50.ty

# 双进程并行（同一模型跑两份，分别在 RC 和 EP）
edge10 --rc-ep flow_infer_ep -m models/onnx_resnet50.ty

edge10 是统一入口，会自动：解析 --rc/--ep/--rc-ep、补全二进制路径、 起 mount namespace（EP）、设置环境变量。详见 09_双卡并行。

4.4 尺寸规则

除 infer 和 track-video 外，所有模块必须显式传 -w -h：

# 错误：ERROR: flow_classify 必须传 -w -h
edge10 --rc flow_classify_rc -m models/onnx_resnet50.ty -i in.jpg -o out.jpg

# 正确
edge10 --rc flow_classify_rc -m models/onnx_resnet50.ty -i in.jpg -o out.jpg -w 1920 -h 1080
  • JPEG / H.264 / H.265 输入：-w -h 是提示，实际尺寸由解码器返回
  • NV12 裸文件：-w -h 必须完全匹配文件真实尺寸

4.5 多文件并行（--rc-ep）

多个输入文件时，--rc-ep 自动走任务队列——RC/EP 两个 worker 抢任务，谁空闲谁跑：

edge10 --rc-ep flow_classify_ep -m models/onnx_resnet50.ty \
    -i "images/*.jpg" -o out/ -w 1920 -h 1080
  • 用户只需记住 --rc-ep，队列逻辑内部处理
  • 输出文件名不带 _rc/_ep，两卡结果合并到一个目录
  • 实测：6 个 3s 视频转码 25.5s → 18.0s（1.42x）

详见 09_双卡并行。
