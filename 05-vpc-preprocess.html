5. VPC 图像预处理

目录

  • 5.1 头文件与句柄
  • 5.2 三个操作
  • 5.3 数据量估算
  • 5.4 端到端示例：NV12 → resize 224 → RGB planar
  • 5.5 零拷贝直写模型输入（推荐生产用法）
  • 5.6 命令行验证

VPC（Vision Pre-Processor）是硬件加速的图像预处理单元，支持 resize、crop、色彩空间转换。所有操作都在 device 内存上完成，不经过 Host。

5.1 头文件与句柄

#include "Edge10/Edge10_vpc.h"

Edge10Vpc* vpc = NULL;
Edge10VpcConfig cfg = { .max_pic_width = 1920, .max_pic_height = 1080, .channel_id = 0 };
Edge10VpcCreate(&vpc, /*device_id=*/0, &cfg);   /* cfg 传 NULL 也可 */
/* ... resize / crop / convert ... */
Edge10VpcDestroy(vpc);

5.2 三个操作

/* resize：把 src 缩放到 dst 的尺寸 */
Edge10Error Edge10VpcResize(Edge10Vpc* vpc, Edge10Tensor* src, Edge10Tensor* dst);

/* crop：从 src 抠出 ROI，写入 dst */
Edge10Rect roi = { .x = 100, .y = 50, .width = 400, .height = 400 };
Edge10Error Edge10VpcCrop(Edge10Vpc* vpc, Edge10Tensor* src, roi, Edge10Tensor* dst);

/* 色彩空间转换：按 src/dst Tensor 的像素格式自动转换 */
Edge10Error Edge10VpcConvert(Edge10Vpc* vpc, Edge10Tensor* src, Edge10Tensor* dst);

常用像素格式（Edge10PixelFormat）

值 宏 说明
1 EDGE10_PIXEL_NV12 YUV420 semi-planar（VDEC 默认输出）
2 EDGE10_PIXEL_NV21
12 EDGE10_PIXEL_RGB888_PACKED RGB 交错
13 EDGE10_PIXEL_BGR888_PACKED BGR 交错
70 EDGE10_PIXEL_RGB888_PLANAR RGB 三平面（模型常用）
71 EDGE10_PIXEL_BGR888_PLANAR BGR 三平面

5.3 数据量估算

  • NV12：width * height * 3 / 2 字节
  • RGB/BGR packed：width * height * 3 字节
  • RGB/BGR planar：同上，按 plane 排布

注意 stride：VDEC / VPC 输出的宽高可能不是 16 对齐的，务必使用 width_stride / height_stride（见 06_硬件编解码）来计算真实 帧大小，不要直接用 width*height*3/2。

5.4 端到端示例：NV12 → resize 224 → RGB planar

完整可跑版本见 samples/src/flow/sample_vpc.c 和 samples/src/single/sample_vpc.c。核心序列：

Edge10Vpc* vpc = NULL;
Edge10VpcCreate(&vpc, 0, NULL);

/* 把 Host 上的 NV12 图上传到 device */
Edge10Buffer* src_dev = NULL;
Edge10Malloc(&src_dev, 0, src_bytes);
void* src_virt = NULL;
Edge10GetVirtAddr(src_dev, &src_virt);
/* memcpy host -> src_virt（H2D） */

/* 构造 src / dst Tensor（dims 仅用于描述形状，实际数据在 device buffer） */
const int64_t src_dims[2]  = { SW, SH };
const int64_t mid_dims[2]  = { TW, TH };   /* 224x224 NV12 */
const int64_t dst_dims[2]  = { 3, TW*TH }; /* RGB planar */
Edge10Tensor *src_t, *mid_t, *dst_t;
Edge10TensorCreate(&src_t, 0, src_dims, 2, EDGE10_DT_UINT8);
Edge10TensorCreate(&mid_t, 0, mid_dims, 2, EDGE10_DT_UINT8);
Edge10TensorCreate(&dst_t, 0, dst_dims, 2, EDGE10_DT_UINT8);

Edge10VpcResize(vpc, src_t, mid_t);       /* 1920x1080 NV12 -> 224x224 NV12 */
Edge10VpcConvert(vpc, mid_t, dst_t);      /* NV12 -> RGB planar */

/* 把 dst_t 拷回 Host 保存 */
Edge10Buffer* dst_buf = NULL; void* dst_virt = NULL;
Edge10TensorGetBuffer(dst_t, &dst_buf);
Edge10GetVirtAddr(dst_buf, &dst_virt);
/* memcpy dst_virt -> host (D2H) */

Edge10TensorDestroy(dst_t); Edge10TensorDestroy(mid_t); Edge10TensorDestroy(src_t);
Edge10Free(src_dev);
Edge10VpcDestroy(vpc);

5.5 零拷贝直写模型输入（推荐生产用法）

推理前通常要把预处理结果直接写进模型的输入 buffer，避免一次额外拷贝。 配合 Edge10ExecutorGetContext：

Edge10Context* ctx = NULL;
Edge10ExecutorGetContext(exec, &ctx);
Edge10Tensor* ctx_in = NULL;
Edge10GetInputTensor(ctx, 0, &ctx_in);
Edge10Buffer* in_buf = NULL;
Edge10TensorGetBuffer(ctx_in, &in_buf);
void* in_virt = NULL; uint64_t in_phy = 0;
Edge10GetVirtAddr(in_buf, &in_virt);
Edge10GetPhysAddr(in_buf, &in_phy);

/* VPC 的 dst 直接指向 in_buf，不再单独分配中间 tensor */
Edge10VpcConvert(vpc, mid_t, ctx_in);

/* 跳过 stage 输入拷贝：传 in=NULL */
Edge10ExecutorInfer(exec, NULL, out_t);

完整例子见 samples/src/flow/sample_pipeline.c 和 samples/src/flow/sample_video_pipeline.c。

5.6 命令行验证

# 把 1920x1080 NV12 缩放到 224x224 并转 RGB planar
edge10 --rc flow_vpc_rc -i /tmp/1080p.nv12 -o /tmp/out.rgb -w 1920 -h 1080
