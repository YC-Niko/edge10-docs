Edge10DeviceInfo

typedef struct {
    int32_t  device_id;
    char     device_name[32];
    char     chip_id[32];
    int32_t  cu_count;
    int32_t  memory_size_mb;
    float    temperature;
    float    voltage;
    uint32_t health;        /* 0=OK, 1=NG */
} Edge10DeviceInfo;

Edge10PcieLinkInfo

typedef struct {
    uint32_t speed_mbps;    /* 2500 / 5000 / 8000 */
    uint32_t width;         /* 1 / 2 / 4 / 8 */
    uint32_t dbdf;          /* domain:bus:dev.func */
    int32_t  online;       /* 1=up, 0=down */
} Edge10PcieLinkInfo;

Edge10Rect

typedef struct {
    int32_t x;
    int32_t y;
    int32_t width;
    int32_t height;
} Edge10Rect;

Edge10InferHandle（异步推理句柄）

typedef struct {
    void*    _internal;
    int32_t  device_id;
} Edge10InferHandle;

不要直接访问 _internal；把整个结构体按值传给 *Query* 即可。

配置结构

typedef struct {
    int32_t max_pic_width;
    int32_t max_pic_height;
    int32_t channel_id;
} Edge10VpcConfig;

typedef struct {
    int32_t width;
    int32_t height;
    int32_t payload_type;   /* 26=JPEG, 96=H.264, 265=H.265 */
    int32_t mode;
    int32_t ref_frame_num;
} Edge10VdecConfig;

typedef struct {
    int32_t width;
    int32_t height;
    int32_t payload_type;
    int32_t rc_mode;
    int32_t bitrate;
    int32_t frame_rate;
    int32_t gop;
    int32_t qp;             /* JPEG 质量 0~100，越大越清晰；默认 80 */
} Edge10VencConfig;
